new: function computeSessionElapsed(s: { time?: { created?: number; completed?: number } }): string {
  const start = s.time?.created
  if (!start) return ""
  const ms = (s.time?.completed ?? Date.now()) - start
  const s2 = Math.floor(ms / 1000)
  if (s2 < 60) return `${s2}s`
  return `${Math.floor(s2 / 60)}m`
}

### Task 2: Add subagent live memos

**File:** `spine-entry.tsx` — after `isAgentEntry` memo (~line 98)

```
old:   // Agent entries (subagent tasks) — always interactive
  const isAgentEntry = createMemo(() => kind() === "agent")
new:   // Agent entries (subagent tasks) — always interactive
  const isAgentEntry = createMemo(() => kind() === "agent")
  // Subagent live status — same pattern as SubagentFooter
  const sync = useSync()
  const subagentSession = createMemo(() => {
    const source = entry().source
    if (!isAgentEntry() || !source?.sessionID) return undefined
    return sync.session.get(source.sessionID)
  })
  const subagentMessages = createMemo(() => {
    const source = entry().source
    if (!isAgentEntry() || !source?.sessionID) return []
    return sync.data.message[source.sessionID] ?? []
  })
  const subagentStatus = createMemo(() => {
    const s = subagentSession()
    if (!s) return undefined
    const done = typeof s.time?.completed === "number" && s.time.completed > 0
    if (done) return { glyph: "◎", label: "done", elapsed: computeSessionElapsed(s) }
    const msgs = subagentMessages()
    if (msgs.length > 0) return { glyph: "◇", label: "running", elapsed: computeSessionElapsed(s) }
    return { glyph: "○", label: "pending", elapsed: computeSessionElapsed(s) }
  })
```

## Confidence: 100%

**Status/elapsed:** `sync.session.get()` is a binary search on `store.session` array — always returns session metadata for ALL sessions (not just active route). `time.created` and `time.completed` are always present → status + elapsed always computable. Zero dependency on message loading.

**Live messages (bonus):** `sync.data.message[childID]` only populated when child session was the active route. When available → shows last message content in `nodeSummary()`. When not available → falls back to static `entry().summary`. Graceful degradation.

**Reactivity:** `sync.session` is a SolidJS store — `subagentSession()` re-runs when any session's metadata changes (e.g., `time.completed` set on completion). Spine entry updates live without polling.
